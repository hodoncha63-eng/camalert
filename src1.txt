package com.example.camalert
import kotlin.math.*
data class Camera(val id: String, val lat: Double, val lng: Double, val limit: Int, val type: String)
data class Hit(val camera: Camera, val distance: Double)
object CameraMatcher {
    private const val R = 6371000.0
    fun distance(lat1: Double, lng1: Double, lat2: Double, lng2: Double): Double {
        val p1 = Math.toRadians(lat1); val p2 = Math.toRadians(lat2)
        val dp = p2 - p1; val dl = Math.toRadians(lng2 - lng1)
        val a = sin(dp / 2).pow(2) + cos(p1) * cos(p2) * sin(dl / 2).pow(2)
        return 2 * R * asin(sqrt(a))
    }
    fun bearing(lat1: Double, lng1: Double, lat2: Double, lng2: Double): Double {
        val p1 = Math.toRadians(lat1); val p2 = Math.toRadians(lat2)
        val dl = Math.toRadians(lng2 - lng1)
        val y = sin(dl) * cos(p2)
        val x = cos(p1) * sin(p2) - sin(p1) * cos(p2) * cos(dl)
        return (Math.toDegrees(atan2(y, x)) + 360) % 360
    }
    private fun angleDiff(a: Double, b: Double): Double {
        val d = abs(a - b) % 360
        return if (d > 180) 360 - d else d
    }
    fun nextAhead(
        cams: List<Camera>, lat: Double, lng: Double, heading: Double,
        maxDist: Double = 600.0, maxAngle: Double = 35.0
    ): Hit? {
        val dLat = maxDist / 111_000.0
        val dLng = maxDist / (111_000.0 * cos(Math.toRadians(lat)))
        var best: Hit? = null
        for (c in cams) {
            if (abs(c.lat - lat) > dLat || abs(c.lng - lng) > dLng) continue
            val d = distance(lat, lng, c.lat, c.lng)
            if (d > maxDist) continue
            if (angleDiff(heading, bearing(lat, lng, c.lat, c.lng)) > maxAngle) continue
            val cur = best
            if (cur == null || d < cur.distance) best = Hit(c, d)
        }
        return best
    }
}

package com.example.camalert
import android.bluetooth.BluetoothDevice
import android.content.BroadcastReceiver
import android.content.Context
import android.content.Intent
import android.os.Build
import androidx.core.content.ContextCompat
class CarConnectionReceiver : BroadcastReceiver() {
    override fun onReceive(c: Context, i: Intent) {
        val dev: BluetoothDevice? =
            if (Build.VERSION.SDK_INT >= 33)
                i.getParcelableExtra(BluetoothDevice.EXTRA_DEVICE, BluetoothDevice::class.java)
            else @Suppress("DEPRECATION") i.getParcelableExtra(BluetoothDevice.EXTRA_DEVICE)
        val saved = c.getSharedPreferences("prefs", Context.MODE_PRIVATE)
            .getString("car_addr", null) ?: return
        if (dev?.address != saved) return
        val svc = Intent(c, AlertService::class.java)
        try {
            when (i.action) {
                BluetoothDevice.ACTION_ACL_CONNECTED -> ContextCompat.startForegroundService(c, svc)
                BluetoothDevice.ACTION_ACL_DISCONNECTED -> c.stopService(svc)
            }
        } catch (e: Exception) {
        }
    }
}

package com.example.camalert
import android.content.Context
import android.speech.tts.TextToSpeech
import android.speech.tts.Voice
import java.util.Locale
object VoiceConfig {
    const val RATE = 0.7f
    const val PITCH = 1.05f
    const val SAMPLE = "전방 오백 미터 앞에 단속카메라가 있어요. 제한속도는 육십 킬로미터예요."
    fun koreanVoices(tts: TextToSpeech): List<Voice> = try {
        val all = tts.voices?.filter { it.locale.language == "ko" } ?: emptyList()
        val offline = all.filter { !it.isNetworkConnectionRequired }
        (if (offline.isNotEmpty()) offline else all).sortedBy { it.name }
    } catch (e: Exception) {
        emptyList()
    }
    fun savedVoiceName(ctx: Context): String? =
        ctx.getSharedPreferences("prefs", Context.MODE_PRIVATE).getString("voice_name", null)
    fun saveVoiceName(ctx: Context, name: String) {
        ctx.getSharedPreferences("prefs", Context.MODE_PRIVATE).edit().putString("voice_name", name).apply()
    }
    private fun defaultFemale(voices: List<Voice>): Voice? =
        voices.firstOrNull { it.name.contains("female", true) }
            ?: voices.firstOrNull { it.name.contains("-kob-") }
            ?: voices.firstOrNull()
    fun apply(tts: TextToSpeech, ctx: Context) {
        tts.language = Locale.KOREAN
        tts.setSpeechRate(RATE)
        tts.setPitch(PITCH)
        val voices = koreanVoices(tts)
        val saved = savedVoiceName(ctx)
        val chosen = voices.firstOrNull { it.name == saved } ?: defaultFemale(voices)
        if (chosen != null) tts.voice = chosen
    }
}
